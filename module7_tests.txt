# idempotency test: a retry of the same logical request must not double-charge Rush's capacity or frequency counters.
def test_successful_allocation():
    store = AllocationStore(redis_client)

    campaign_id = 100001
    viewer_id = "viewer-success"
    request_id = "request-success"

    decision_time = datetime(
        2026,
        10,
        5,
        21,
        30,
        tzinfo=timezone.utc,
    )

    capacity_key = store._remaining_capacity_key(campaign_id)
    frequency_key = store._frequency_key(
        campaign_id,
        viewer_id,
        decision_time,
    )
    decision_key = store._decision_key(request_id)

    # Start with clean Redis state.
    redis_client.delete(
        capacity_key,
        frequency_key,
        decision_key,
    )

    # Campaign starts with 5 remaining allocations.
    store.initialize_new_campaign_capacity(
        campaign_id,
        5,
    )

    result = store.try_allocate(
        campaign_id=campaign_id,
        viewer_id=viewer_id,
        request_id=request_id,
        frequency_cap_per_hour=3,
        decision_time=decision_time,
    )

    # The allocation should succeed.
    assert result == AllocationResult.ALLOCATED

    # One unit of capacity should have been consumed.
    assert redis_client.get(capacity_key) == "4"

    # Viewer should now have seen this campaign once this hour.
    assert redis_client.get(frequency_key) == "1"

    # The request should remember which campaign was allocated.
    assert redis_client.get(decision_key) == str(campaign_id)

    # Cleanup.
    redis_client.delete(
        capacity_key,
        frequency_key,
        decision_key,
    )

def test_frequency_cap_rejects_allocation():
    store = AllocationStore(redis_client)

    campaign_id = 100005
    viewer_id = "viewer-frequency-capped"
    request_id = "request-frequency-capped"

    decision_time = datetime(
        2026,
        10,
        5,
        21,
        30,
        tzinfo=timezone.utc,
    )

    capacity_key = store._remaining_capacity_key(campaign_id)
    frequency_key = store._frequency_key(
        campaign_id,
        viewer_id,
        decision_time,
    )
    decision_key = store._decision_key(request_id)

    redis_client.delete(
        capacity_key,
        frequency_key,
        decision_key,
    )

    store.initialize_new_campaign_capacity(
        campaign_id,
        5,
    )

    # Viewer has already hit the hourly cap.
    redis_client.set(

def test_concurrent_allocations_do_not_overallocate():
    store = AllocationStore(redis_client)

    campaign_id = 100008
    viewer_id = "viewer-concurrency"

    decision_time = datetime(
        2026,
        10,
        5,
        21,
        30,
        tzinfo=timezone.utc,
    )

    capacity_key = store._remaining_capacity_key(campaign_id)
    frequency_key = store._frequency_key(
        campaign_id,
        viewer_id,
        decision_time,
    )

    request_1 = "request-concurrent-1"
    request_2 = "request-concurrent-2"

    decision_key_1 = store._decision_key(request_1)
    decision_key_2 = store._decision_key(request_2)

    redis_client.delete(
        capacity_key,
        frequency_key,
        decision_key_1,
        decision_key_2,
    )

    # Only ONE allocation is available.
    store.initialize_new_campaign_capacity(
        campaign_id,
        1,
    )

    def allocate(request_id: str):
        return store.try_allocate(
            campaign_id=campaign_id,
            viewer_id=viewer_id,
            request_id=request_id,
            frequency_cap_per_hour=10,
            decision_time=decision_time,
        )

    # Fire both allocation attempts concurrently.
    with ThreadPoolExecutor(max_workers=2) as executor:
        results = list(
            executor.map(
                allocate,
                [request_1, request_2],
            )
        )

    # Exactly one request must win.
    assert results.count(AllocationResult.ALLOCATED) == 1
    assert results.count(AllocationResult.CAPACITY_EXHAUSTED) == 1

    # Capacity must stop at zero — never negative.
    assert redis_client.get(capacity_key) == "0"

    # Only the winning allocation should increment frequency.
    assert redis_client.get(frequency_key) == "1"

    redis_client.delete(
        capacity_key,
        frequency_key,
        decision_key_1,
        decision_key_2,
    )

def test_allocation_version_fence():
    # Use our local Redis instance
    client = Redis(host="localhost", port=6379, decode_responses=True)
    store = AllocationStore(client)

    # Use unique identifiers to avoid existing test data
    test_id = uuid4().hex
    campaign_id = 999999
    request_id = f"version-fence-test-{test_id}"
    viewer_id = f"version-fence-viewer-{test_id}"

    decision_time = datetime.now(timezone.utc)

    capacity_key = store._remaining_capacity_key(campaign_id)
    frequency_key = store._frequency_key(
        campaign_id, viewer_id, decision_time
    )
    decision_key = store._decision_key(request_id)
    version_key = "rush:index:version"

    # Save the existing global version; don't change it
    current_version = client.get(version_key)
    assert current_version is not None, "Global index version is missing"

    current_version = int(current_version)

    try:
        # Prepare a campaign with 100 units remaining
        client.set(capacity_key, 100)

        # Viewer has already seen this campaign once
        client.set(frequency_key, 1, ex=3600)

        # Make sure this request has not been processed
        client.delete(decision_key)

        # Simulate a worker using an outdated index version
        stale_version = current_version - 1

        result = store.try_allocate(
            campaign_id=campaign_id,
            request_id=request_id,
            viewer_id=viewer_id,
            frequency_cap_per_hour=3,
            decision_time=decision_time,
            index_version=stale_version,
        )

        # Redis must reject the outdated snapshot
        assert result == AllocationResult.STALE_INDEX

        # No capacity should have been consumed
        assert store.get_remaining_capacity(campaign_id) == 100

        # Frequency count must remain unchanged
        assert store.get_frequency_count(
            campaign_id, viewer_id, decision_time
        ) == 1

        # No idempotency key should have been created
        assert client.get(decision_key) is None

        print("Lua version fence test passed!")

    finally:
        # Remove only the test's dedicated keys
        client.delete(capacity_key, frequency_key, decision_key)

def test_create_and_get_campaign():
    # Create a campaign
    test_input_campaign = {
        "name": "Nike",
        "capacity": 62,
        "frequency_cap_per_hour": 3,
        "start_time": "2026-10-10T18:00:00",
        "end_time": "2026-10-25T23:00:00",
    }

    create_response = client.post(
        "/campaigns",
        json=test_input_campaign,
    )

    assert create_response.status_code == 201

    create_response_body = create_response.json()

    # Use whatever ID PostgreSQL generated
    campaign_id = create_response_body["id"]

    # Fetch the same campaign using that ID
    get_response = client.get(
        f"/campaigns/{campaign_id}"
    )

    assert get_response.status_code == 200

    get_response_body = get_response.json()

    # Confirm we got back the same campaign

        "regions": ["Mumbai"],
        "devices": ["MOBILE"],
        "subscription_tiers": ["PREMIUM"],
    }

    first_response = client.post(
        "/campaigns",
        json={
            **campaign_input,
            "name": "Exhausted Campaign",
        },
    )

    second_response = client.post(
        "/campaigns",
        json={
            **campaign_input,
            "name": "Healthy Campaign",
        },
    )

    first_id = first_response.json()["id"]
    second_id = second_response.json()["id"]

    allocation_store = AllocationStore(redis_client)

    redis_client.delete(
        allocation_store._remaining_capacity_key(first_id),
        allocation_store._remaining_capacity_key(second_id),
    )

    client.post(f"/campaigns/{first_id}/activate")
    client.post(f"/campaigns/{second_id}/activate")

    # First candidate has no remaining capacity
    redis_client.set(
        allocation_store._remaining_capacity_key(first_id),
        0,
    )

    first_campaign = db_session.get(Campaign, first_id)
    second_campaign = db_session.get(Campaign, second_id)

    request_id = f"fallback-test-{uuid4()}"

    # Force deterministic candidate order:
    # exhausted campaign first, healthy campaign second.
    with patch.object(
        CampaignRepository,
        "find_candidates",
        return_value=[first_campaign, second_campaign],
    ):
        response = client.post(
            "/decisions",
            json={
                "request_id": request_id,
                "viewer_id": f"viewer-{uuid4()}",
                "region": "Mumbai",
                "device": "MOBILE",
                "subscription_tier": "PREMIUM",
            },
        )

    assert response.status_code == 200

    body = response.json()

    assert body["status"] == "ALLOCATED"
    assert body["campaign_id"] == second_id

    assert allocation_store.get_remaining_capacity(first_id) == 0
    assert allocation_store.get_remaining_capacity(second_id) == 99
from unittest.mock import MagicMock, patch

from app.runtime.allocation_store import AllocationResult
from app.services.decision_service import make_decision


def test_stale_index_triggers_retry():
    # Simulate a campaign being discovered after a refresh.
    campaign_12 = MagicMock()
    campaign_12.id = 12
    campaign_12.frequency_cap_per_hour = 3

    campaign_15 = MagicMock()
    campaign_15.id = 15
    campaign_15.frequency_cap_per_hour = 3

    # No actual database or index needed.
    db = MagicMock()
    index = MagicMock()
    repository = MagicMock()
    allocation_store = MagicMock()

    # First discovery uses version 2.
    # Second discovery uses version 3 with different candidates.
    index.find_candidates.side_effect = [
        ({12}, 2),
        ({15}, 3),
    ]

    repository.get_candidates_by_ids.side_effect = [
        [campaign_12],
        [campaign_15],
    ]

    # First allocation is rejected as stale.
    # Second allocation succeeds.
    allocation_store.try_allocate.side_effect = [
        AllocationResult.STALE_INDEX,
        AllocationResult.ALLOCATED,
    ]

    allocation_store.get_existing_decision.return_value = None

    with (
        patch(
            "app.services.decision_service.CampaignRepository",
            return_value=repository,
        ),
        patch(
            "app.services.decision_service.AllocationStore",
            return_value=allocation_store,
        ),
        patch(
            "app.services.decision_service.ensure_index_current",
            return_value=True,
        ) as refresh_mock,
    ):
        request = MagicMock()
        request.request_id = "retry-test-001"
        request.viewer_id = "viewer-001"
        request.region = "Mumbai"
        request.device = "Mobile"
        request.subscription_tier = "Premium"

        result = make_decision(request, db, index)

    assert result is campaign_15

    # Freshness was checked on both attempts.
    assert refresh_mock.call_count == 2

    # Candidate discovery happened twice.
    assert index.find_candidates.call_count == 2

    # PostgreSQL candidate retrieval happened twice.
    assert repository.get_candidates_by_ids.call_count == 2

    # Allocation was attempted twice.
    assert allocation_store.try_allocate.call_count == 2

    # Verify we did not retry with the original campaign.
    assert allocation_store.try_allocate.call_args_list[0].args[0] == 12
    assert allocation_store.try_allocate.call_args_list[0].args[5] == 2

    assert allocation_store.try_allocate.call_args_list[1].args[0] == 15
    assert allocation_store.try_allocate.call_args_list[1].args[5] == 3

    print("Stale index refresh-and-retry test passed!")


def test_retry_is_bounded():
    db = MagicMock()
    index = MagicMock()
    repository = MagicMock()
    allocation_store = MagicMock()

    campaign = MagicMock()
    campaign.id = 12

        ({12}, 2),

    allocation_store.try_allocate.return_value = AllocationResult.STALE_INDEX

    with (
        patch(
            "app.services.decision_service.CampaignRepository",

            "app.services.decision_service.AllocationStore",

from app.db.database import SessionLocal
from app.repositories.campaign_repository import CampaignRepository
from app.services.inverted_index import InvertedIndex
from app.models.campaign import Campaign
from datetime import datetime, timedelta, timezone

def test_load_index():
    db = SessionLocal()
    try:
        repository = CampaignRepository(db)
        index = InvertedIndex()
        now = datetime.now(timezone.utc)
        start_time = now - timedelta(hours=1)
        end_time = now + timedelta(hours=1)
        # Next: load campaign data from PostgreSQL
        campaign=Campaign(
            # id will be automatically generated by postgreSQL when repository.add() calls flush()
            name="Haldirams Dussherra Campaign",
            capacity=1000,
            status="ACTIVE",
            frequency_cap_per_hour=3,
            start_time=start_time,
            end_time=end_time
        )
        repository.add(campaign)
        # To add targeting values to the session
        repository.add_targeting(
            campaign,
            regions=["Mumbai"],
            devices=["Mobile"],
            subscription_tiers=["Premium"]
        )
        
        # To send the pending INSERTs to Postgres
        db.flush()
        # To add all the active campaigns to the in-memory index
        repository.load_index(index)
        
        list_of_IDs=index.find_candidates("Mumbai", "Mobile", "Premium")
        assert campaign.id in list_of_IDs, "Campaign ID is NOT part of the index"
        
        # Mismatch test
        non_matching_ids = index.find_candidates("Delhi", "Mobile", "Premium")
        assert campaign.id not in non_matching_ids, "Campaign incorrectly matched Delhi"
        
        # Retrieve Campaign objects using the indexed IDs
        campaign_objects = repository.get_candidates_by_ids(
            list_of_IDs,
            now #datetime
        )
        
        # Extract IDs from the returned Campaign objects
        returned_ids = []
        print("Campaign objects:", campaign_objects)

        for campaign_obj in campaign_objects:
            # Append the campaign object's ID to returned_ids
            returned_ids.append(campaign_obj.id)

        # Assert that our test campaign's ID is present
        assert campaign.id in returned_ids
        
        # Test a decision made before the campaign starts
        before_start = start_time - timedelta(minutes=1)

        early_candidates = repository.get_candidates_by_ids(
                list_of_IDs,
                before_start
            )

        assert campaign.id not in {c.id for c in early_candidates}

        # Test a decision made after the campaign expires
        after_end = end_time + timedelta(minutes=1)

        expired_candidates = repository.get_candidates_by_ids(
                list_of_IDs,
                after_end
            )

        assert campaign.id not in {c.id for c in expired_candidates}



    finally:
        db.rollback()
        db.close()
        
        
if __name__ == "__main__":
    test_load_index()
    print("PostgreSQL index integration test passed!")
